-- Run this once in Supabase SQL Editor to create the TMS table
create table if not exists public.tms_trips (
  id text primary key,
  shift_date text not null,
  expectable_time text not null,
  route text not null,
  route_group text not null,
  team text not null,
  staff_names text not null,
  address text not null,
  created_at bigint not null,
  done_time text,
  done_timestamp bigint,
  duration_ms bigint,
  status text not null default 'In Transit',
  is_grouped boolean default false
);

-- Enable open read/write policies for staff & admin devices
alter table public.tms_trips enable row level security;

drop policy if exists "Allow public read tms_trips" on public.tms_trips;
create policy "Allow public read tms_trips" on public.tms_trips for select using (true);

drop policy if exists "Allow public insert tms_trips" on public.tms_trips;
create policy "Allow public insert tms_trips" on public.tms_trips for insert with check (true);

drop policy if exists "Allow public update tms_trips" on public.tms_trips;
create policy "Allow public update tms_trips" on public.tms_trips for update using (true);

drop policy if exists "Allow public delete tms_trips" on public.tms_trips;
create policy "Allow public delete tms_trips" on public.tms_trips for delete using (true);

-- Enable Real-Time updates across devices
alter publication supabase_realtime add table public.tms_trips;
